using UnityEngine; using UnityEngine.EventSystems; using System.Collections.Generic; using System.IO; // ======================================================== // 1. بيانات الحفظ للتصدير إلى JSON // ======================================================== [System.Serializable] public class BlockData { public int blockType; // 0: Block, 1: Wheel, 2: Thruster public Vector3 localPosition; public Quaternion localRotation; } [System.Serializable] public class VehicleSaveData { public List blocks = new List(); } // ======================================================== // 2. مكون الصاروخ (Thruster) // ======================================================== public class ThrusterComponent : MonoBehaviour { public ParticleSystem thrustParticles; public void SetActiveEffect(bool active) { if (thrustParticles == null) return; if (active && !thrustParticles.isPlaying) thrustParticles.Play(); else if (!active && thrustParticles.isPlaying) thrustParticles.Stop(); } } // ======================================================== // 3. النظام الشامل للبناء والقيادة والتأثيرات والحفظ // ======================================================== public class VehicleBuilderSystem : MonoBehaviour { public enum BuildObjectType { Block, Wheel, Thruster } public enum BuildToolMode { Place, Delete } [Header("القطع المتاحة (Prefabs)")] public GameObject blockPrefab; public GameObject wheelPrefab; public GameObject thrusterPrefab; [Header("مواد المعاينة الشفافة (Preview Materials)")] public Material previewValidMaterial; [Header("إعدادات الشبكة والطبقات")] public LayerMask buildableLayer; public float gridSize = 1.0f; [Header("إعدادات الفيزياء والحركة")] public float wheelSpeed = 1000f; public float wheelTorque = 800f; public float thrusterForce = 2500f; public float steeringSpeedMultiplier = 0.6f; [Header("إعدادات الكاميرا")] public Transform cameraTransform; public Vector3 cameraOffset = new Vector3(0, 5, -10); public float cameraSmoothSpeed = 5f; [Header("حالة اللعبة الحالية")] public bool isPlayMode = false; public BuildToolMode currentToolMode = BuildToolMode.Place; [Header("مدخلات الجوال (Mobile Controls)")] [Range(-1f, 1f)] public float mobileMoveInput = 0f; [Range(-1f, 1f)] public float mobileSteerInput = 0f; public bool isMobileThrusterActive = false; private BuildObjectType currentSelectedType = BuildObjectType.Block; private GameObject coreBlock; private Camera mainCam; private GameObject previewObject; private Vector2 touchStartPosition; private float touchStartTime; private readonly List spawnedBlocks = new List(); void Start() { mainCam = Camera.main; if (cameraTransform == null && mainCam != null) cameraTransform = mainCam.transform; SetPhysicsState(false); UpdatePreviewObject(); } void Update() { if (Input.GetKeyDown(KeyCode.Space)) TogglePlayMode(); if (!isPlayMode) { if (Input.GetKeyDown(KeyCode.Alpha1)) SelectBlockType(0); if (Input.GetKeyDown(KeyCode.Alpha2)) SelectBlockType(1); if (Input.GetKeyDown(KeyCode.Alpha3)) SelectBlockType(2); HandlePreview(); HandleInput(); } else { if (previewObject != null) previewObject.SetActive(false); } } void FixedUpdate() { if (isPlayMode) { HandleWheelMovement(); HandleThrusterForce(); UpdateCameraFollow(); } } void HandlePreview() { if (previewObject == null) UpdatePreviewObject(); Vector2 mousePos = Input.mousePosition; if (Input.touchCount > 0) mousePos = Input.GetTouch(0).position; Ray ray = mainCam.ScreenPointToRay(mousePos); if (Physics.Raycast(ray, out RaycastHit hit, 100f, buildableLayer) && !IsPointerOverUI()) { previewObject.SetActive(currentToolMode == BuildToolMode.Place); Vector3 spawnPosition = SnapToGrid(hit.point + hit.normal * (gridSize * 0.5f)); Quaternion spawnRotation = Quaternion.FromToRotation(Vector3.up, hit.normal); previewObject.transform.position = spawnPosition; previewObject.transform.rotation = spawnRotation; } else { previewObject.SetActive(false); } } void UpdatePreviewObject() { if (previewObject != null) Destroy(previewObject); GameObject prefab = GetSelectedPrefab(); if (prefab == null) return; previewObject = Instantiate(prefab); previewObject.name = "Build_Preview_Ghost"; foreach (Collider col in previewObject.GetComponentsInChildren()) Destroy(col); foreach (Rigidbody rb in previewObject.GetComponentsInChildren()) Destroy(rb); if (previewValidMaterial != null) { foreach (Renderer rend in previewObject.GetComponentsInChildren()) rend.material = previewValidMaterial; } previewObject.SetActive(false); } void HandleInput() { if (Input.touchCount == 1) { Touch touch = Input.GetTouch(0); if (IsPointerOverUI(touch.fingerId)) return; if (touch.phase == TouchPhase.Began) { touchStartPosition = touch.position; touchStartTime = Time.time; } else if (touch.phase == TouchPhase.Ended) { float touchDuration = Time.time - touchStartTime; float touchDistance = Vector2.Distance(touch.position, touchStartPosition); if (touchDuration < 0.3f && touchDistance < 20f) { if (currentToolMode == BuildToolMode.Place) PlaceBlockAtPosition(touch.position); else DeleteBlockAtPosition(touch.position); } } return; } if (Input.GetMouseButtonDown(0) && !IsPointerOverUI()) { if (currentToolMode == BuildToolMode.Place) PlaceBlockAtPosition(Input.mousePosition); else DeleteBlockAtPosition(Input.mousePosition); } else if (Input.GetMouseButtonDown(1) && !IsPointerOverUI()) { DeleteBlockAtPosition(Input.mousePosition); } } bool IsPointerOverUI(int fingerId = -1) { if (EventSystem.current == null) return false; if (fingerId >= 0) return EventSystem.current.IsPointerOverGameObject(fingerId); return EventSystem.current.IsPointerOverGameObject(); } void PlaceBlockAtPosition(Vector2 screenPosition) { if (mainCam == null) return; Ray ray = mainCam.ScreenPointToRay(screenPosition); if (Physics.Raycast(ray, out RaycastHit hit, 100f, buildableLayer)) { Vector3 spawnPosition = SnapToGrid(hit.point + hit.normal * (gridSize * 0.5f)); Quaternion spawnRotation = Quaternion.FromToRotation(Vector3.up, hit.normal); GameObject prefabToSpawn = GetSelectedPrefab(); if (prefabToSpawn == null) return; GameObject newBlock = Instantiate(prefabToSpawn, spawnPosition, spawnRotation); spawnedBlocks.Add(newBlock); Rigidbody newRb = newBlock.GetComponent(); if (newRb == null) newRb = newBlock.AddComponent(); newRb.isKinematic = !isPlayMode; if (currentSelectedType == BuildObjectType.Thruster && newBlock.GetComponent() == null) newBlock.AddComponent(); if (coreBlock == null) coreBlock = newBlock; if (hit.collider.attachedRigidbody != null && hit.collider.gameObject != newBlock) { if (currentSelectedType == BuildObjectType.Block || currentSelectedType == BuildObjectType.Thruster) { FixedJoint joint = newBlock.AddComponent(); joint.connectedBody = hit.collider.attachedRigidbody; } else if (currentSelectedType == BuildObjectType.Wheel) { HingeJoint hinge = newBlock.GetComponent(); if (hinge == null) hinge = newBlock.AddComponent(); hinge.connectedBody = hit.collider.attachedRigidbody; } } } } void DeleteBlockAtPosition(Vector2 screenPosition) { if (mainCam == null) return; Ray ray = mainCam.ScreenPointToRay(screenPosition); if (Physics.Raycast(ray, out RaycastHit hit, 100f)) { GameObject target = hit.collider.gameObject; if (spawnedBlocks.Contains(target)) { if (target == coreBlock) coreBlock = null; spawnedBlocks.Remove(target); Destroy(target); spawnedBlocks.RemoveAll(item => item == null); if (coreBlock == null && spawnedBlocks.Count > 0) coreBlock = spawnedBlocks[0]; } } } public void SaveVehicle(string filename = "vehicle.json") { if (coreBlock == null || spawnedBlocks.Count == 0) return; VehicleSaveData saveData = new VehicleSaveData(); foreach (GameObject block in spawnedBlocks) { if (block == null) continue; BlockData data = new BlockData { localPosition = coreBlock.transform.InverseTransformPoint(block.transform.position), localRotation = Quaternion.Inverse(coreBlock.transform.rotation) * block.transform.rotation }; if (block.GetComponent() != null) data.blockType = 1; else if (block.GetComponent() != null) data.blockType = 2; else data.blockType = 0; saveData.blocks.Add(data); } string json = JsonUtility.ToJson(saveData, true); string path = Path.Combine(Application.persistentDataPath, filename); File.WriteAllText(path, json); Debug.Log("تم الحفظ بنجاح في: " + path); } public void LoadVehicle(string filename = "vehicle.json") { string path = Path.Combine(Application.persistentDataPath, filename); if (!File.Exists(path)) return; ClearVehicle(); string json = File.ReadAllText(path); VehicleSaveData saveData = JsonUtility.FromJson(json); if (saveData.blocks.Count == 0) return; Vector3 basePos = transform.position; for (int i = 0; i < saveData.blocks.Count; i++) { BlockData data = saveData.blocks[i]; GameObject prefab = (data.blockType == 1) ? wheelPrefab : (data.blockType == 2) ? thrusterPrefab : blockPrefab; Vector3 spawnPos = basePos + data.localPosition; GameObject newBlock = Instantiate(prefab, spawnPos, data.localRotation); spawnedBlocks.Add(newBlock); Rigidbody rb = newBlock.GetComponent(); if (rb == null) rb = newBlock.AddComponent(); rb.isKinematic = !isPlayMode; if (data.blockType == 2 && newBlock.GetComponent() == null) newBlock.AddComponent(); if (i == 0) coreBlock = newBlock; } RebuildJoints(); } void RebuildJoints() { for (int i = 1; i < spawnedBlocks.Count; i++) { GameObject block = spawnedBlocks[i]; if (block == null) continue; if (block.GetComponent() == null && block.GetComponent() == null) { FixedJoint joint = block.AddComponent(); joint.connectedBody = coreBlock.GetComponent(); } else if (block.GetComponent() != null) { HingeJoint hinge = block.GetComponent(); hinge.connectedBody = coreBlock.GetComponent(); } } } public void ClearVehicle() { foreach (GameObject block in spawnedBlocks) if (block != null) Destroy(block); spawnedBlocks.Clear(); coreBlock = null; } public void SelectBlockType(int typeIndex) { currentSelectedType = (BuildObjectType)typeIndex; currentToolMode = BuildToolMode.Place; UpdatePreviewObject(); } public void SetToolMode(int modeIndex) => currentToolMode = (BuildToolMode)modeIndex; public void TogglePlayMode() { isPlayMode = !isPlayMode; SetPhysicsState(isPlayMode); } public void SetMobileMoveInput(float value) => mobileMoveInput = value; public void SetMobileSteerInput(float value) => mobileSteerInput = value; public void SetMobileThruster(bool active) => isMobileThrusterActive = active; void SetPhysicsState(bool enablePhysics) { foreach (GameObject block in spawnedBlocks) { if (block != null) { Rigidbody rb = block.GetComponent(); if (rb != null) rb.isKinematic = !enablePhysics; } } } void HandleWheelMovement() { float moveInput = Input.GetAxis("Vertical"); if (Mathf.Abs(mobileMoveInput) > 0.05f) moveInput = mobileMoveInput; float steerInput = Input.GetAxis("Horizontal"); if (Mathf.Abs(mobileSteerInput) > 0.05f) steerInput = mobileSteerInput; foreach (GameObject block in spawnedBlocks) { if (block != null) { HingeJoint hinge = block.GetComponent(); if (hinge != null) { JointMotor motor = hinge.motor; motor.maxMotorTorque = wheelTorque; if (Mathf.Abs(moveInput) > 0.05f || Mathf.Abs(steerInput) > 0.05f) { float localX = 0f; if (coreBlock != null) { Vector3 localPos = coreBlock.transform.InverseTransformPoint(block.transform.position); localX = localPos.x; } float steerModifier = (localX >= 0) ? -steerInput : steerInput; float calculatedSpeed = (moveInput * wheelSpeed) + (steerModifier * wheelSpeed * steeringSpeedMultiplier); motor.targetVelocity = calculatedSpeed; hinge.motor = motor; hinge.useMotor = true; } else { motor.targetVelocity = 0f; hinge.motor = motor; hinge.useMotor = true; } } } } } void HandleThrusterForce() { bool isThrusting = Input.GetKey(KeyCode.LeftShift) || isMobileThrusterActive; foreach (GameObject block in spawnedBlocks) { if (block != null) { ThrusterComponent thruster = block.GetComponent(); if (thruster != null) { thruster.SetActiveEffect(isThrusting); if (isThrusting) { Rigidbody rb = block.GetComponent(); if (rb != null) rb.AddForce(block.transform.forward * thrusterForce, ForceMode.Force); } } } } } void UpdateCameraFollow() { if (coreBlock != null && cameraTransform != null) { Vector3 targetPosition = coreBlock.transform.position + cameraOffset; cameraTransform.position = Vector3.Lerp(cameraTransform.position, targetPosition, cameraSmoothSpeed * Time.fixedDeltaTime); cameraTransform.LookAt(coreBlock.transform.position); } } GameObject GetSelectedPrefab() { switch (currentSelectedType) { case BuildObjectType.Block: return blockPrefab; case BuildObjectType.Wheel: return wheelPrefab; case BuildObjectType.Thruster: return thrusterPrefab; default: return blockPrefab; } } Vector3 SnapToGrid(Vector3 position) { return new Vector3( Mathf.Round(position.x / gridSize) * gridSize, Mathf.Round(position.y / gridSize) * gridSize, Mathf.Round(position.z / gridSize) * gridSize ); } } // ======================================================== // 4. عصا التحكم بالجوال (Virtual Joystick) // ======================================================== public class VirtualJoystick : MonoBehaviour, IDragHandler, IPointerDownHandler, IPointerUpHandler { [Header("عناصر الـ UI")] public RectTransform joystickBackground; public RectTransform joystickHandle; [Header("إعدادات الحركة")] public float handleRange = 1f; [Header("ربط نظام Builder System")] public VehicleBuilderSystem vehicleBuilderSystem; private Vector2 inputVector = Vector2.zero; void Start() { if (joystickBackground == null) joystickBackground = GetComponent(); if (vehicleBuilderSystem == null) vehicleBuilderSystem = FindObjectOfType(); } public void OnDrag(PointerEventData eventData) { Vector2 bgSize = joystickBackground.rect.size; if (RectTransformUtility.ScreenPointToLocalPointInRectangle( joystickBackground, eventData.position, eventData.pressEventCamera, out Vector2 pos)) { pos.x = (pos.x / bgSize.x) * 2f; pos.y = (pos.y / bgSize.y) * 2f; inputVector = new Vector2(pos.x, pos.y); inputVector = (inputVector.magnitude > 1.0f) ? inputVector.normalized : inputVector; if (joystickHandle != null) { joystickHandle.anchoredPosition = new Vector2( inputVector.x * (bgSize.x / 2f) * handleRange, inputVector.y * (bgSize.y / 2f) * handleRange ); } SendInputToVehicle(); } } public void OnPointerDown(PointerEventData eventData) => OnDrag(eventData); public void OnPointerUp(PointerEventData eventData) { inputVector = Vector2.zero; if (joystickHandle != null) joystickHandle.anchoredPosition = Vector2.zero; SendInputToVehicle(); } void SendInputToVehicle() { if (vehicleBuilderSystem != null) { vehicleBuilderSystem.SetMobileSteerInput(inputVector.x); vehicleBuilderSystem.SetMobileMoveInput(inputVector.y); } } } // ======================================================== // 5. كاميرا البناء والتحويط (Build Camera Controller) // ======================================================== public class BuildCameraController : MonoBehaviour { [Header("الهدف والإعدادات الأساسية")] public Transform target; public float distance = 10.0f; [Header("سرعة الدوران")] public float xSpeed = 120.0f; public float ySpeed = 120.0f; public float yMinLimit = -20f; public float yMaxLimit = 80f; [Header("إعدادات التقريب (Zoom)")] public float zoomSpeed = 2f; public float minDistance = 3f; public float maxDistance = 20f; private float x = 0.0f; private float y = 0.0f; void Start() { Vector3 angles = transform.eulerAngles; x = angles.y; y = angles.x; } void LateUpdate() { if (target == null) return; if ((Input.GetMouseButton(1) || Input.GetMouseButton(2)) && !IsPointerOverUI()) { x += Input.GetAxis("Mouse X") * xSpeed * 0.02f; y -= Input.GetAxis("Mouse Y") * ySpeed * 0.02f; } float scroll = Input.GetAxis("Mouse ScrollWheel"); if (Mathf.Abs(scroll) > 0.01f) { distance = Mathf.Clamp(distance - scroll * zoomSpeed * 5f, minDistance, maxDistance); } HandleTouchControls(); y = Mathf.Clamp(y, yMinLimit, yMaxLimit); Quaternion rotation = Quaternion.Euler(y, x, 0); Vector3 position = rotation * new Vector3(0.0f, 0.0f, -distance) + target.position; transform.rotation = rotation; transform.position = position; } void HandleTouchControls() { if (Input.touchCount == 1) { Touch touch = Input.GetTouch(0); if (touch.phase == TouchPhase.Moved && !IsPointerOverUI(touch.fingerId)) { x += touch.deltaPosition.x * xSpeed * 0.005f; y -= touch.deltaPosition.y * ySpeed * 0.005f; } } else if (Input.touchCount == 2) { Touch touchZero = Input.GetTouch(0); Touch touchOne = Input.GetTouch(1); Vector2 touchZeroPrevPos = touchZero.position - touchZero.deltaPosition; Vector2 touchOnePrevPos = touchOne.position - touchOne.deltaPosition; float prevTouchDeltaMag = (touchZeroPrevPos - touchOnePrevPos).magnitude; float touchDeltaMag = (touchZero.position - touchOne.position).magnitude; float deltaMagnitudeDiff = prevTouchDeltaMag - touchDeltaMag; distance = Mathf.Clamp(distance + deltaMagnitudeDiff * zoomSpeed * 0.01f, minDistance, maxDistance); } } bool IsPointerOverUI(int fingerId = -1) { if (EventSystem.current == null) return false; if (fingerId >= 0) return EventSystem.current.IsPointerOverGameObject(fingerId); return EventSystem.current.IsPointerOverGameObject(); } }
